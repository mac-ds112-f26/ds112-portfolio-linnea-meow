# Side-by-side boxplots (defined below)
ggplot(elections, aes(y = repub_pct_20, x = historical)) +
  geom_boxplot()

  • top of line: maximum

  • top of box: 75th percentile

  • line in box: median

  • bottom of box: 25th percentile

  • bottom of line: minimum

  • dots outside: outlier(s)

  • historically blue states’ median vote was slightly over 50% republican in 2020, as were purple states (but moreso). historically red states’ median vote across counties was around 75% republican.

-the largest range/diversity of voting was in blue states

#exercise 9: quant vs. categorical - density plots

Code
# Name two "bad" things about this plot - 1. you can't see the values hidden by the fill 2. the y-axis doesn't provide a communicable scale/reference point - unclear what high density means here
ggplot(elections, aes(x = repub_pct_20, fill = historical)) +
  geom_density()
Code
# What does scale_fill_manual do? - allows you to choose the colors manually
ggplot(elections, aes(x = repub_pct_20, fill = historical)) +
  geom_density() +
  scale_fill_manual(values = c("blue", "purple", "red"))
Code
# What does alpha = 0.5 do? - changes opacity level! you can now see the hidden values beneath the other colors
# Play around with different values of alpha, between 0 and 1
ggplot(elections, aes(x = repub_pct_20, fill = historical)) +
  geom_density(alpha = 0.1) +
  scale_fill_manual(values = c("blue", "purple", "red"))
Code
# What does facet_wrap do?! - separates the density plots as 3 separate graphics
ggplot(elections, aes(x = repub_pct_20, fill = historical)) +
  geom_density() +
  scale_fill_manual(values = c("blue", "purple", "red")) +
  facet_wrap(~historical)
Code
# Let's try a similar grouping strategy with a histogram instead of density plot.
# Why is this terrible? - you can't see the values underneath, and alpha doesn't help!!
ggplot(elections, aes(x = repub_pct_20, fill = historical)) +
  geom_histogram(color = "white", alpha = 0.1) +
  scale_fill_manual(values = c("blue", "purple", "red"))

#exercise 10 - reflect - i like the density plot with opacity fill because it’s visually pleasing, but box plots are more informative - density plots show relative scale of each trend - box plots show the character of the data (its median, its range, its outliers)

library(ggridges)

# THINK: still hard to read because of how many states there are; scale on small R window looks ugly; reverse ABC order

  geom_density_ridges()
Code
# Let's put the states in order by Republican support, not alphabet
# How do you think fct_reorder works? We'll learn about this later in the semester. - factors reordered based on its relation to another value
ggplot(elections, aes(x = repub_pct_20, y = fct_reorder(state_abbr, repub_pct_20))) + 
  geom_density_ridges(alpha = 0.5)
Code
# YOUR TURN: color/fill the ridges according to a state's historical voting patterns

  ylab("State") + 
  scale_fill_manual(values = c("blue", "purple", "red"))

  • least variability: HI, NH, TN, IN, WV
